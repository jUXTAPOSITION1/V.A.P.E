```python
import requests
from typing import Optional

class HTTPAgent:
    def __init__(self, url: str, timeout: int = 10):
        """
        Initialize the HTTP agent with a URL and optional timeout.
        
        :param url: The URL to interact with
        :param timeout: The request timeout in seconds (default: 10)
        """
        self.url = url
        self.timeout = timeout

    def get(self) -> Optional[str]:
        """
        Send a GET request to the URL and return the response text.
        
        :return: The response text or None if an error occurs
        """
        try:
            response = requests.get(self.url, timeout=self.timeout)
            response.raise_for_status()
            return response.text
        except requests.RequestException as e:
            logging.error(f"Error sending GET request: {e}")
            return None
```
