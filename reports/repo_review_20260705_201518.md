import re

def _addr(req):
    if isinstance(req, dict):
        for k in ("address", "contract", "token", "target"):
            if req.get(k):
                addr = str(req[k]).strip()
                if re.match(r"^0x[a-fA-F0-9]{40}$", addr):
                    return addr
    if isinstance(req, str):
        m = re.search(r"0x[a-fA-F0-9]{40}", req)
